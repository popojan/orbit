\begin{abstract}
We construct a canonical Fibonacci representation for rational numbers: every $p/q$ can be written as $(\sum_i F_{a_i})/F_n$, where $n = z(q)$ is the Fibonacci entry point and $\{a_i\}$ is the Zeckendorf decomposition of $p \cdot (F_n/q)$. This extends classical Zeckendorf representation from integers to all rationals. The construction yields a polynomial form $P(\varphi)/Q(\varphi) = p/q$ via Binet's formula, where $\sqrt{5}$ factors cancel due to the anti-symmetric structure $\varphi^k - \psi^k$. We analyze the complexity of this representation: the expected number of Zeckendorf terms satisfies $\mathbb{E}[k] = z(q)/(1+\varphi^2) + O(1)$, where the constant $1/(1+\varphi^2) = (5-\sqrt{5})/10$ is Lekkerkerker's density. We also develop arithmetic operations: the sum of two Fibonacci fractions has denominator $F_N$ where $N = \mathrm{lcm}(z(q_1), z(q_2))$. While Zeckendorf arithmetic achieves optimal complexity---$O(n)$ for addition, $O(n \log n)$ for multiplication---the practical utility for general rationals is limited by the computational cost of finding entry points.
\end{abstract}

\section{Introduction}

Zeckendorf's theorem \cite{zeckendorf1972} states that every positive integer has a unique representation as a sum of non-consecutive Fibonacci numbers. The Fibonacci entry point (or rank of apparition) $z(q)$ is the smallest positive integer $n$ such that $q$ divides $F_n$---a value guaranteed to exist by the Pisano period property \cite{OEIS-A001175}.

Freitag and Filipponi \cite{freitag1990,filipponi1993} studied Zeckendorf representations of specific Fibonacci ratios $F_{kn}/F_n$. We generalize this to \emph{all} rational numbers by combining the entry point with Zeckendorf decomposition.

\textbf{Main contributions:}
\begin{enumerate}
    \item Canonical Fibonacci form for all rationals $\mathbb{Q}$
    \item Polynomial representation via Binet substitution
    \item Complexity analysis via Lekkerkerker's theorem
    \item Arithmetic operations with complexity analysis
    \item Comparison with Egyptian fraction and continued fraction representations
\end{enumerate}

\section{Preliminaries}

\begin{definition}[Fibonacci Entry Point]
For $q \in \mathbb{Z}^+$, the entry point $z(q)$ is the smallest $n \geq 1$ such that $q \mid F_n$.
\end{definition}

\begin{theorem}[Pisano]
For every $q \in \mathbb{Z}^+$, the entry point $z(q)$ exists and divides the Pisano period $\pi(q)$.
\end{theorem}

\begin{definition}[Zeckendorf Representation]
For $n \in \mathbb{Z}^+$, the Zeckendorf representation is the unique set of indices $\{a_1 > a_2 > \cdots > a_k\}$ with $a_i - a_{i+1} \geq 2$ such that $n = \sum_i F_{a_i}$.
\end{definition}

\section{Main Results}

\subsection{The Fibonacci Fraction Construction}

\begin{construction}[Canonical Fibonacci Form]
\label{con:main}
For any rational $p/q > 0$, define the \emph{Fibonacci form} as:
\[
\frac{p}{q} = \frac{\sum_{i} F_{a_i}}{F_n}
\]
where:
\begin{itemize}
    \item $n = z(q)$ is the Fibonacci entry point of $q$
    \item $\{a_i\} = \mathrm{Zeck}(m)$ where $m = p \cdot F_n/q$ is the \emph{scaled numerator}
\end{itemize}
\end{construction}

\begin{proposition}[Well-definedness]
\label{prop:welldef}
The Fibonacci form is well-defined for all positive rationals.
\end{proposition}

\begin{proof}
By Pisano periodicity, $z(q)$ exists for every $q \in \mathbb{Z}^+$, so $q \mid F_n$ and $F_n/q \in \mathbb{Z}$. Let $m = p \cdot (F_n/q) \in \mathbb{Z}^+$. Then:
\[
\frac{p}{q} = \frac{p \cdot (F_n/q)}{F_n} = \frac{m}{F_n} = \frac{\sum_i F_{a_i}}{F_n}
\]
where the last equality applies Zeckendorf's theorem to $m$.
\end{proof}

\begin{remark}[Why $z(q)$ is canonical]
\label{rem:canonical}
One might ask: why use $n = z(q)$ rather than any multiple $kz(q)$? Since $q \mid F_{z(q)}$ implies $q \mid F_{kz(q)}$ for all $k \geq 1$, infinitely many choices work.

The entry point $z(q)$ is canonical because it is \emph{minimal}: it yields the lowest-degree polynomial representation. Using $n = 2z(q)$ would double the polynomial degree without adding information. This parallels the convention of using reduced fractions $p/q$ with $\gcd(p,q) = 1$---any multiple $(kp)/(kq)$ represents the same rational, but the reduced form is canonical.
\end{remark}

\begin{remark}
The construction works for any $p/q$, not just reduced fractions. For non-reduced $p'/q' = p/q$, the entry point $z(q')$ may differ from $z(q)$, but the final rational value is the same.
\end{remark}

\begin{remark}[Why not Zeck$(p)$/Zeck$(q)$?]
\label{rem:whynot}
One might ask: why not simply represent $p/q$ as $\mathrm{Zeck}(p)/\mathrm{Zeck}(q)$, i.e., take Zeckendorf representations of numerator and denominator separately?

This fails because $(\sum_i F_{a_i})/(\sum_j F_{b_j})$ does not generally equal $p/q$ when evaluated at $\varphi$. The Zeckendorf representation of an integer $n$ satisfies $n = \sum F_{a_i}$, but at $\varphi$:
\[
\sum_i \varphi^{a_i} \neq n \quad \text{(in general)}
\]

Our construction works because:
\begin{enumerate}
    \item The denominator is a \emph{single} Fibonacci number $F_n$
    \item The entry point ensures $q \mid F_n$, so $F_n/q \in \mathbb{Z}$
    \item The $\sqrt{5}$ factors cancel in $(\varphi^{a_i} - \psi^{a_i})/(\varphi^n - \psi^n)$
\end{enumerate}
The single-Fibonacci denominator is essential for the Galois-theoretic cancellation.
\end{remark}

\begin{corollary}[GCD Property]
For reduced $p/q$:
\[
\gcd\left(\sum_i F_{a_i}, F_n\right) = \frac{F_n}{q}
\]
\end{corollary}

\begin{proof}
Let $m = \sum_i F_{a_i} = p \cdot (F_n/q)$ and $d = F_n/q$. Then $\gcd(m, F_n) = \gcd(pd, qd) = d \cdot \gcd(p, q) = d$ since $\gcd(p,q) = 1$.
\end{proof}

\subsection{Polynomial Form}

\begin{proposition}[$\varphi$-Polynomial Representation]
\label{prop:poly}
Every rational $p/q$ can be expressed as $P(\varphi)/Q(\varphi)$ where $P(x), Q(x) \in \mathbb{Z}[x]$ are derived from the Fibonacci form via Binet's formula. The polynomials are determined by the Zeckendorf indices.
\end{proposition}

\textbf{Construction.} Using Binet's formula $F_n = (\varphi^n - \psi^n)/\sqrt{5}$ where $\psi = 1 - \varphi$:
\[
\frac{p}{q} = \frac{\sum_i (\varphi^{a_i} - \psi^{a_i})}{\varphi^n - \psi^n}
\]
The $\sqrt{5}$ factors cancel. Define the polynomial form by the formal substitution $x \mapsto \varphi$, $(1-x) \mapsto \psi$ in $\mathbb{Z}[x]$:
\[
\frac{P(x)}{Q(x)} = \frac{\sum_i (x^{a_i} - (1-x)^{a_i})}{x^n - (1-x)^n}
\]
Note: this is a formal substitution; in $\mathbb{R}$ we have $\psi = (1-\sqrt{5})/2 \neq 1 - \varphi$, but the algebraic identity $\varphi + \psi = 1$ justifies the correspondence.

\subsection{Special Point $x = 1/2$}

\begin{observation}
At $x = 1/2$, we have $x = 1-x$, so every term $x^n - (1-x)^n = 0$. Both $P(1/2) = 0$ and $Q(1/2) = 0$, giving a removable singularity. Since each factor $x^k - (1-x)^k$ has a simple zero at $x = 1/2$, both $P$ and $Q$ factor as $(x - 1/2) \cdot \tilde{P}(x)$ and $(x - 1/2) \cdot \tilde{Q}(x)$ respectively, so $\lim_{x \to 1/2} P(x)/Q(x) = \tilde{P}(1/2)/\tilde{Q}(1/2)$ exists.
\end{observation}

Note that $P(\varphi)/Q(\varphi) = p/q$ and $P(\psi)/Q(\psi) = p/q$ (by Galois invariance), but the limit at $x = 1/2$ is generally \emph{not} equal to $p/q$.

\section{Algebraic Structure}

The representation theorem has a natural interpretation in terms of the algebraic number field $\mathbb{Q}(\sqrt{5})$.

\subsection{The Ring $\mathbb{Z}[\varphi]$}

The golden ratio $\varphi = (1+\sqrt{5})/2$ is an algebraic integer satisfying $\varphi^2 = \varphi + 1$. The ring of integers in $\mathbb{Q}(\sqrt{5})$ is:
\[
\mathbb{Z}[\varphi] = \{a + b\varphi : a, b \in \mathbb{Z}\}
\]

Every power of $\varphi$ reduces to this form via the recurrence $\varphi^n = F_n \cdot \varphi + F_{n-1}$, where $F_n$ is the $n$-th Fibonacci number.

\subsection{Galois Conjugation}

The field $\mathbb{Q}(\sqrt{5})$ has Galois group $\mathrm{Gal}(\mathbb{Q}(\sqrt{5})/\mathbb{Q}) = \{1, \sigma\}$ where the conjugation $\sigma$ acts by:
\[
\sigma(\varphi) = \psi = 1 - \varphi = \frac{1-\sqrt{5}}{2}
\]

The key Galois-theoretic operations are:
\begin{itemize}
    \item \textbf{Trace:} $\mathrm{Tr}(\varphi^n) = \varphi^n + \psi^n = L_n$ (Lucas numbers)
    \item \textbf{Norm:} $\mathrm{N}(\varphi^n) = \varphi^n \cdot \psi^n = (\varphi\psi)^n = (-1)^n$
\end{itemize}

\subsection{Why $\sqrt{5}$ Cancels}

The polynomial $f(x, k) = x^k - (1-x)^k$ satisfies the \emph{anti-symmetry property}:
\[
f(1-x, k) = -f(x, k)
\]

Under Galois conjugation $\sigma: \varphi \mapsto \psi = 1-\varphi$:
\[
f(\varphi, k) = \varphi^k - \psi^k = F_k \cdot \sqrt{5}
\]

This is precisely the Binet formula. In our rational representation:
\[
\frac{P(\varphi)}{Q(\varphi)} = \frac{\sum_i (\varphi^{a_i} - \psi^{a_i})}{\varphi^n - \psi^n} = \frac{\sum_i F_{a_i} \cdot \sqrt{5}}{F_n \cdot \sqrt{5}} = \frac{\sum_i F_{a_i}}{F_n} \in \mathbb{Q}
\]

The $\sqrt{5}$ factors cancel because both numerator and denominator use the \emph{anti-symmetric} combination $\varphi^k - \psi^k$, which always produces $F_k \cdot \sqrt{5}$.

\begin{theorem}[Galois Invariance]
For any $p/q \in \mathbb{Q}$, the polynomial representation satisfies:
\[
\frac{P(\varphi)}{Q(\varphi)} = \frac{P(\psi)}{Q(\psi)} = \frac{p}{q}
\]
That is, the value is fixed by Galois conjugation, confirming it lies in $\mathbb{Q}$.
\end{theorem}

\subsection{Fibonacci vs Lucas}

The Fibonacci and Lucas sequences arise from complementary Galois operations:

\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
Sequence & Formula & Galois Property \\
\midrule
$F_n$ & $(\varphi^n - \psi^n)/\sqrt{5}$ & Anti-symmetric (involves $\sqrt{5}$) \\
$L_n$ & $\varphi^n + \psi^n$ & Symmetric (trace, no $\sqrt{5}$) \\
\bottomrule
\end{tabular}
\end{center}

A ``Lucas fraction'' representation would use $g(x,k) = x^k + (1-x)^k$, which is symmetric: $g(1-x,k) = g(x,k)$. However, Lucas numbers lack a universal entry point property: not every integer divides some Lucas number.

\begin{observation}
The Fibonacci representation works precisely because:
\begin{enumerate}
    \item Every $q$ divides some $F_n$ (Pisano period property)
    \item The anti-symmetric form $\varphi^k - \psi^k$ produces $\sqrt{5}$, which cancels
    \item The result is Galois-invariant, hence rational
\end{enumerate}
\end{observation}

\section{Complexity Analysis}

\begin{definition}[Fibonacci Complexity]
The complexity $C(p/q)$ of a rational is the pair $(n, k)$ where $n = z(q)$ is the entry point and $k$ is the number of Zeckendorf terms.
\end{definition}

\subsection{Expected Number of Terms}

The key question is: how does $k$ depend on the entry point $n = z(q)$?

\begin{theorem}[Lekkerkerker's Density, 1951]
\label{thm:lekkerkerker}
For integers $m$ in the range $[F_j, F_{j+1})$, the average number of summands in the Zeckendorf representation is:
\[
\mathbb{E}[|\mathrm{Zeck}(m)|] = \frac{j}{1 + \varphi^2} + O(1) = \frac{j}{\varphi + 2} + O(1)
\]
where the constant $1/(1+\varphi^2) = (5-\sqrt{5})/10 \approx 0.2764$ is called \emph{Lekkerkerker's constant}.
\end{theorem}

\begin{corollary}[Fibonacci Fraction Complexity]
\label{cor:complexity}
Fix $q$ with entry point $n = z(q)$, and let $d = F_n/q$. For reduced fractions $p/q$ with $p \in \{1, \ldots, q-1\}$, the scaled numerator $m = pd$ ranges over $\{d, 2d, \ldots, (q-1)d\}$. The average number of Zeckendorf terms over this family is:
\[
\bar{k} = \frac{n}{1 + \varphi^2} + O(1) \approx 0.2764 \cdot n
\]
\end{corollary}

\begin{proof}
The values $\{d, 2d, \ldots, (q-1)d\}$ are evenly spaced in $(0, F_n)$ with spacing $d$. Since Lekkerkerker's theorem gives the average over \emph{all} integers in $[F_{n-1}, F_n)$, and our sample is a regular sublattice, the same asymptotic density applies.
\end{proof}

\textbf{Numerical verification.} Using Farey sequences $\mathcal{F}_N$ for systematic sampling:
\begin{itemize}
    \item Linear regression on 3043 fractions: slope $= 0.2754$, $R^2 = 0.9999$ (aggregated by $q$)
    \item Theoretical prediction: $(5-\sqrt{5})/10 \approx 0.2764$
    \item Individual variance: $R^2 \approx 0.92$ (due to combinatorial fluctuation in Zeckendorf)
\end{itemize}

\begin{remark}[Statistical vs Exact]
Fibonacci complexity is inherently \emph{statistical} due to Zeckendorf combinatorics: the number of summands has variance $\approx 0.95$ and converges to a Gaussian distribution \cite{kologlu2010}.

In contrast, Egyptian fraction complexity is \emph{exact}: the author shows in \cite{egypt2025} that $|\mathrm{Egypt}(p/q)| = \lceil |\mathrm{CF}(p/q)|/2 \rceil$, where each pair of CF coefficients determines one Egyptian tuple via a telescoping sum.
\end{remark}

\subsection{Comparison with Other Representations}

\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
Representation & Complexity Formula & Governs \\
\midrule
Continued fraction & $|\mathrm{CF}(p/q)|$ & Approximation quality \\
Egyptian fraction & $\lceil |\mathrm{CF}|/2 \rceil$ & CF structure (exact) \\
Fibonacci fraction & $z(q)/(1+\varphi^2) + O(1)$ & Pisano entry point (statistical) \\
\bottomrule
\end{tabular}
\end{center}

The three complexities measure different aspects:
\begin{itemize}
    \item \textbf{CF}: How well does $p/q$ approximate nearby irrationals?
    \item \textbf{Egyptian}: Derived from CF via algebraic transformation
    \item \textbf{Fibonacci}: How deep into Pisano sequence before $q \mid F_n$?
\end{itemize}

\begin{observation}
Primes $q$ with large $z(q)$ yield high Fibonacci complexity. For Mersenne primes $M_p = 2^p - 1$, the entry point satisfies $z(M_p) \in \{M_p - 1, M_p + 1\}$ (cf.\ \cite{wall1960}). Since $z(M_p) \approx M_p$, these give near-maximal complexity.
\end{observation}

\begin{remark}[Coding-theoretic application]
Zeckendorf representations yield \emph{self-synchronizing codes}: the non-consecutive constraint ensures no codeword is a prefix of another, enabling error recovery without delimiters \cite{apostolico1987}. The Fibonacci fraction representation inherits this property---the Zeckendorf indices $\{a_i\}$ form a prefix-free encoding of the numerator. This may be useful for transmitting rationals in noisy channels.
\end{remark}

\subsection{Cyclotomic Divisibility}

The sixth cyclotomic polynomial $\Phi_6(x) = 1 - x + x^2$ plays a special role because $\Phi_6(\varphi) = 1 - \varphi + \varphi^2 = 2$.

\begin{theorem}[Cyclotomic Divisibility]
For the polynomial form $P(x)/Q(x)$ of $p/q$:
\begin{enumerate}
\item $\Phi_6(x) \mid Q(x) \iff z(q) \equiv 0 \pmod{3}$
\item $\Phi_6(x) \mid P(x) \iff \#\{a_i \equiv 1,2 \pmod{6}\} = \#\{a_i \equiv 4,5 \pmod{6}\}$
\end{enumerate}
where $\{a_i\}$ are the Zeckendorf indices of the scaled numerator.
\end{theorem}

\begin{proof}
Let $\omega = e^{i\pi/3}$ be a primitive 6th root of unity, so $\omega^2 - \omega + 1 = 0$ and $1 - \omega = \bar{\omega}$.

For $Q(x) = x^n - (1-x)^n$: we have $Q(\omega) = \omega^n - \bar{\omega}^n = 2i\sin(n\pi/3)$, which vanishes iff $n \equiv 0 \pmod{3}$.

For $P(x) = \sum_i (x^{a_i} - (1-x)^{a_i})$: we have $P(\omega) = 2i\sum_i \sin(a_i\pi/3)$. Since $\sin(n\pi/3)$ equals $\sqrt{3}/2$ for $n \equiv 1,2 \pmod{6}$, equals $-\sqrt{3}/2$ for $n \equiv 4,5 \pmod{6}$, and vanishes for $n \equiv 0,3 \pmod{6}$, the sum $P(\omega) = 0$ iff the ``positive'' and ``negative'' index counts balance.
\end{proof}

\begin{remark}
When $\Phi_6$ divides exactly one of $P$ or $Q$, it contributes a factor of $2$ to the numerator or denominator at $x = \varphi$. The frequency of $\Phi_6 \mid Q$ (condition: $3 \mid z(q)$) depends on the distribution of Fibonacci entry points. The frequency of $\Phi_6 \mid P$ (the balance condition) decreases as $q$ grows, since longer Zeckendorf sequences are less likely to achieve exact mod-6 balance.
\end{remark}

\section{Arithmetic Operations}

A natural question is how to perform arithmetic directly on Fibonacci fraction representations. We show that addition has clean structure, while the practical utility is limited by computational bottlenecks.

\subsection{Addition Theorem}

\begin{theorem}[Fibonacci Fraction Addition]
\label{thm:addition}
For rationals $r_1 = p_1/q_1$ and $r_2 = p_2/q_2$ with Fibonacci entry points $n_1 = z(q_1)$ and $n_2 = z(q_2)$, the sum $r_1 + r_2$ has Fibonacci denominator $F_N$ where:
\[
N = \mathrm{lcm}(n_1, n_2)
\]
\end{theorem}

\begin{proof}
Since $q_1 \mid F_{n_1}$ and $q_2 \mid F_{n_2}$, we need a common Fibonacci divisor. By the divisibility property $F_n \mid F_m \iff n \mid m$, the smallest $F_N$ divisible by both $F_{n_1}$ and $F_{n_2}$ has $N = \mathrm{lcm}(n_1, n_2)$.

The common denominator $\mathrm{lcm}(q_1, q_2)$ divides $F_N$, so we can scale both fractions:
\[
r_1 + r_2 = \frac{m_1 \cdot (F_N/F_{n_1}) + m_2 \cdot (F_N/F_{n_2})}{F_N}
\]
where $m_i$ are the scaled numerators. The result follows by applying Zeckendorf to the sum.
\end{proof}

\subsection{Index Scaling via Lucas Numbers}

The scaling factor $F_N/F_n$ when $N = kn$ can be computed using Lucas numbers:

\begin{lemma}[Fibonacci-Lucas Product Identity]
\[
F_a \cdot L_n = F_{a+n} + (-1)^{n+1} F_{a-n}
\]
In particular, $F_{2n}/F_n = L_n$.
\end{lemma}

This identity transforms Zeckendorf indices: when scaling by $L_n$, each index $a$ maps to $\{a+n, a-n\}$ with appropriate sign, producing a signed intermediate sum that must be normalized.

\subsection{Complexity of Zeckendorf Arithmetic}

The algorithmic complexity of arithmetic in Zeckendorf representation has been studied extensively:

\begin{theorem}[Ahlbach, Usatine, Pippenger 2012 \cite{ahlbach2012}]
Addition and subtraction of $n$-digit Zeckendorf numbers can be performed in $O(n)$ time using three alternating passes. These operations can also be implemented by combinational circuits of size $O(n)$ and depth $O(\log n)$.
\end{theorem}

\begin{theorem}[Idziaszek 2021 \cite{idziaszek2021}]
Multiplication of $n$-digit Zeckendorf numbers can be performed in $O(n \log n)$ time via conversion to the golden ratio base $\varphi$, FFT convolution, and normalization.
\end{theorem}

The key insight of Idziaszek is that the Zeckendorf system converts in $O(n)$ time to/from the base-$\varphi$ positional system, where FFT-based multiplication applies directly.

\subsection{The Entry Point Bottleneck}

\begin{observation}[Practical Limitation]
While Zeckendorf arithmetic achieves optimal complexity---$O(n)$ for addition, $O(n \log n)$ for multiplication, matching binary arithmetic---the \emph{Fibonacci fraction} arithmetic faces a bottleneck: computing the entry point $z(q)$.

For general $q$, computing $z(q)$ requires knowing the prime factorization of $q$, since:
\begin{itemize}
    \item $z(p)$ for prime $p$ divides $p^2 - 1$ (or $2(p^2-1)$ if $p \equiv \pm 1 \pmod{5}$)
    \item $z(\prod p_i^{e_i}) = \mathrm{lcm}(z(p_i^{e_i}))$
\end{itemize}
No polynomial-time algorithm is known for computing $z(q)$ without factoring.
\end{observation}

\begin{remark}[When Fibonacci Arithmetic is Efficient]
Fibonacci fraction arithmetic becomes competitive when:
\begin{enumerate}
    \item Denominators are Fibonacci numbers: $z(F_n) = n$ trivially
    \item Entry points are precomputed or cached
    \item Working within a ``Fibonacci lattice'' where all denominators divide some fixed $F_N$
\end{enumerate}
In the special case of Fibonacci-denominator fractions, all operations achieve $O(n \log n)$ complexity, competitive with the best rational arithmetic algorithms, with the structural advantage that no large-integer GCD computation is needed.
\end{remark}

\section{Approximation of Irrationals}

\textbf{Algorithm} (FibonacciRationalize). Given $x \in \mathbb{R}$ and accuracy $\varepsilon$: iterate $n = 3, 4, \ldots$, compute $m = \mathrm{Round}(x \cdot F_n)$, return when $|m/F_n - x| < \varepsilon$.

\section{Related Work}

Zeckendorf \cite{zeckendorf1972} established the unique representation of positive integers as sums of non-consecutive Fibonacci numbers. Freitag and Filipponi \cite{freitag1990,filipponi1993} studied Zeckendorf representations of specific Fibonacci ratios $F_{kn}/F_n$, showing these have predictable structure. Our construction extends their work from this subfamily to all rationals $\mathbb{Q}$.

The divisibility properties underlying our construction---Pisano periods and entry points---were studied by Wall \cite{wall1960} and later by Cubre and Rouse \cite{cubre2012}. Bergman \cite{bergman1957} introduced the golden ratio base (phinary), a related positional system using powers of $\varphi$; our polynomial representation can be viewed as a rational-function analogue.

\section{Open Questions}

\begin{enumerate}
    \item \textbf{Wall's conjecture connection}: Wall's conjecture (1960, still open) states $z(p^2) = p \cdot z(p)$ for odd primes $p$. Does the Fibonacci representation reveal structure related to this conjecture? For primes where $z(p^2) \neq p \cdot z(p)$ (if any exist), what happens to the representation?

    \item \textbf{Minimal representations}: Are there alternative Fibonacci fraction representations with fewer terms than the canonical form?

    \item \textbf{Complexity bounds}: Can we characterize rationals with bounded Fibonacci complexity? (Note: $k = 1$ is achieved exactly when $p/q = F_i/F_j$ for some Fibonacci numbers---a corollary of the construction.)

    \item \textbf{Lucas extension}: Lucas numbers lack a universal entry point property (not every integer divides some $L_n$). Is there a partial Lucas representation for rationals whose denominators do divide some Lucas number?

    \item \textbf{Egyptian-Fibonacci relationship}: Egyptian and Fibonacci representations both express rationals as sums. Egyptian complexity is $\lceil |\mathrm{CF}|/2 \rceil$ (exact), while Fibonacci is $z(q)/(1+\varphi^2)$ (statistical). Is there a deeper structural connection between these representations?

    \item \textbf{Higher cyclotomic factors}: The Cyclotomic Divisibility Theorem characterizes when $\Phi_6(x)$ divides $P(x)$ or $Q(x)$. Do similar characterizations exist for other cyclotomic polynomials?

    \item \textbf{Polynomial factorization}: What number-theoretic properties of $p/q$ are reflected in the factorization structure of $P(x)/Q(x)$ beyond the cyclotomic divisibility conditions?
\end{enumerate}
