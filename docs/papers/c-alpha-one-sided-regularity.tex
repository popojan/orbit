\title{One-sided regularity of the Beatty ballot constant:\\
the path-count identification, left jumps, right flatness,\\
and a closed form for the jump}
\author{Jan Popelka\thanks{Email: popojan@protonmail.com}}
\date{June 2026}

\begin{abstract}
Let $a_\alpha(n)$ count monotone lattice paths from $(1,0)$ to $(n,n)$
that stay weakly below the line $y = \alpha x$, for a slope
$\alpha > 1$, and let $\sigma(\alpha)$ be the survival probability of
the associated fair-coin gap walk.
We first prove the identification
$a_\alpha(n) \sim \sigma(\alpha)\,\binom{2n-1}{n-1}$
(Theorem~\ref{thm:zero}; previously verified numerically in the
companion survey~\cite{bridge}), so that the ballot constant
$C(\alpha) = \sigma(\alpha)/2$ exists for \emph{every} real
$\alpha > 1$.
We also prove that the periodic boundary system of~\cite{bridge} is
nonsingular and computes $\sigma(p/q)$
(Proposition~\ref{prop:nonsing}).
On this basis we establish: at every rational $p/q$ the left limit
$C^-(p/q)$ exists and equals the constant of the same periodic walk
with a sharpened absorbing barrier, giving an \emph{algebraic closed
form} for the jump $J(p/q) = C(p/q) - C^-(p/q)$; the function $C$ is
flat to infinite order from the right at every rational, with an
explicit exponential bound; and $C$ is continuous at every irrational,
with a stretched-exponential modulus governed by the Diophantine type.
For integer slopes the closed form collapses to
$1 - 2\,C^-(k) = \rho_k^{\,k}$, where $\rho_k$ is the ruin probability
of the $\{+k, -1\}$ walk.
Finally, we prove that $C$ is a \emph{pure jump function} ---
$C(\beta) - C(\alpha) = \sum_{r \in (\alpha,\beta]} J(r)$, by a
Dirichlet-covering argument riding on the uniform flatness --- and
that the jumps decay at the exact exponential rate
$J(p/q) = \mu(p/q)^{\,q(1+o(1))}$ with
$\mu(\alpha) = (\alpha+1)^{\alpha+1}/(2^{\alpha+1}\alpha^{\alpha})$,
the Cram\'er tilt rate of the column chain.
\end{abstract}

\section{Setup}

Fix a slope $\alpha > 1$.
A monotone path from $(1,0)$ to $(n,n)$ consists of $n-1$ right-steps
($R$) and $n$ up-steps ($U$); it stays \emph{under the barrier} if
$y \leq \alpha x$ at every lattice point visited, equivalently if the
integer \emph{gap}
\begin{equation}\label{eq:gap}
s_t \;=\; \floor{\alpha(\tau_t + 1)} - u_t
\end{equation}
is nonnegative for all $t$, where $\tau_t$ and $u_t$ count the
right-steps and up-steps among the first $t$ steps.
Let $a_\alpha(n)$ be the number of such paths; the unconstrained count
is $a_\infty(n) = \binom{2n-1}{n-1} \sim 4^n/(2\sqrt{\pi n})$.

The same gap process driven by an infinite sequence of independent
fair coin flips ($R$ or $U$, probability $\tfrac12$ each) is the
\emph{free walk}; it \emph{survives} if $s_t \geq 0$ for all $t$, and
we write $\sigma(\alpha)$ for its survival probability.
For rational $\alpha = p/q$ in lowest terms the rise sequence
$r_j = \floor{\alpha(j+1)} - \floor{\alpha j}$ is $q$-periodic; the
ruin probabilities $\rho(s, j)$ from height $s$ in phase
$j = x \bmod q$ satisfy the periodic recursion of~\cite{bridge}, and
$\sigma(p/q) = 1 - \rho(s_0, j_0)$ with $s_0 = \floor{p/q}$,
$j_0 = 1$.

Two elementary observations are used repeatedly.
First, since $\floor{x} > x - 1$,
\begin{equation}\label{eq:linbound}
s_t \;>\; \alpha\,\tau_t - u_t + (\alpha - 1) .
\end{equation}
Second, monotonicity: if $\alpha' \leq \alpha''$ then
$\floor{\alpha' m} \leq \floor{\alpha'' m}$ for all $m$, so under the
common coin-flip coupling the gap processes satisfy
$s'_t \leq s''_t$ pointwise and $\sigma(\alpha') \leq \sigma(\alpha'')$.

\section{The path-count identification}

\begin{lemma}[Prefix comparison]\label{lem:prefix}
Let $P_n$ denote the uniform measure on the $\binom{2n-1}{n-1}$
unconstrained paths and $P$ the fair-coin measure.
For every event $A$ determined by the first $T$ steps, with
$T \leq n^{1/2}$,
\[
\bigl|P_n(A) - P(A)\bigr| \;\leq\; c_0\,\frac{T^2}{n}
\]
for an absolute constant $c_0$.
\end{lemma}

\begin{proof}
Set $m = 2n-1$, $k = n-1$.
A fixed prefix of $T$ steps containing $r$ right-steps has
$P_n$-probability $\binom{m-T}{k-r}/\binom{m}{k}$ and $P$-probability
$2^{-T}$.
Their ratio is
\[
2^{T}\,\frac{\binom{m-T}{k-r}}{\binom{m}{k}}
= 2^{T}\,
\frac{\prod_{i=0}^{r-1}(k-i)\;\prod_{j=0}^{T-r-1}(m-k-j)}
     {\prod_{l=0}^{T-1}(m-l)}
= \prod_{\text{$T$ factors}}
\frac{1 - O(T/n)}{1 - O(T/n)}
= 1 + O\!\left(\frac{T^2}{n}\right),
\]
uniformly over $0 \leq r \leq T$, since every numerator factor lies in
$[\,n - T,\, n\,]$ and every denominator factor in
$[\,2n - 1 - T,\, 2n - 1\,]$, while $2^T$ matches the $T$ factors of
$\approx \tfrac12$.
Summing over the atoms of $A$ gives the claim.
\end{proof}

\begin{lemma}[Late ruin under the bridge]\label{lem:bridgelate}
Fix $\alpha > 1$.
There are constants $c = c(\alpha) > 0$, $T_0 = T_0(\alpha)$ such that
for all $n$ and all $T \geq T_0$,
\[
P_n\bigl(\exists\, t \in (T,\, 2n-1]:\ s_t < 0\bigr)
\;\leq\; \frac{e^{-cT}}{1 - e^{-c}} .
\]
\end{lemma}

\begin{proof}
By~\eqref{eq:linbound}, $s_t < 0$ forces
$\tau_t < (t - \alpha + 1)/(1 + \alpha) \leq t/(1+\alpha)$.
Under $P_n$ the count $\tau_t$ is hypergeometric (the number of
right-steps in a sample of size $t$ drawn without replacement from
$n-1$ rights and $n$ ups), with mean
$t\,\frac{n-1}{2n-1} \geq \frac{t}{2} - \frac{1}{2}$.
By Hoeffding's inequality for sampling without
replacement~\cite{hoeffding}, $\tau_t$ obeys the binomial Chernoff
bound:
\[
P_n\Bigl(\tau_t \leq \tfrac{t}{1+\alpha}\Bigr)
\;\leq\; \exp\bigl(-2 t\,(\varepsilon_\alpha - O(1/t))^2\bigr),
\qquad
\varepsilon_\alpha = \tfrac12 - \tfrac{1}{1+\alpha}
= \tfrac{\alpha - 1}{2(\alpha+1)} > 0 .
\]
Choose $T_0$ so that the exponent is $\leq -ct$ for $t \geq T_0$ with
$c = \varepsilon_\alpha^2$, and sum over $t > T$.
\end{proof}

\begin{theorem}[Path-count identification]\label{thm:zero}
For every real $\alpha > 1$,
\[
\lim_{n \to \infty} \frac{a_\alpha(n)}{a_\infty(n)} = \sigma(\alpha),
\qquad\text{with}\qquad
\Bigl|\frac{a_\alpha(n)}{a_\infty(n)} - \sigma(\alpha)\Bigr|
= O\!\left(\frac{\log^2 n}{n}\right).
\]
Consequently $a_\alpha(n) \sim C(\alpha)\, 4^n/\sqrt{\pi n}$ with
$C(\alpha) = \sigma(\alpha)/2$.
\end{theorem}

\begin{proof}
$a_\alpha(n)/a_\infty(n) = P_n(\text{survival})$.
Let $A_T = \{s_t \geq 0 \text{ for } t \leq T\}$, a prefix event.
Then
\[
\bigl|P_n(\text{surv}) - \sigma\bigr|
\leq \bigl|P_n(A_T) - P(A_T)\bigr|
+ P_n\bigl(A_T \setminus \text{surv}\bigr)
+ P\bigl(A_T \setminus \{\text{surv. forever}\}\bigr).
\]
The first term is $O(T^2/n)$ by Lemma~\ref{lem:prefix}; the second is
$\leq B e^{-cT}$ by Lemma~\ref{lem:bridgelate}; the third is
$\leq B' \lambda^{T}$ by the free late-contact bound
(Lemma~\ref{lem:chernoff} below, applied with contact level $-1$).
Taking $T = \lceil K \log n \rceil$ with $K$ large enough that
$e^{-cT} + \lambda^T = O(1/n)$ makes all three terms
$O(\log^2 n / n)$.
\end{proof}

\begin{remark}
The $O(\log^2 n / n)$ rate is consistent with the empirical
$1/n$-convergence of the normalized counts
$a_\alpha(n)\sqrt{\pi n}/4^n$ used by the numerical estimators
in~\cite{bridge}.
Theorem~\ref{thm:zero} requires no rationality: it defines
$C(\alpha)$ for every real slope, so the regularity statements below
concern a genuine function on $(1, \infty)$.
\end{remark}

\begin{remark}[The finite-size constant]\label{rem:finitesize}
The true error is asymptotically $c_1(\alpha)/n$, and the constant is
explicit: expanding the prefix weight of Lemma~\ref{lem:prefix} to
first order,
$2^T\binom{m-T}{k-r}/\binom{m}{k} = 1 + \bigl(\tfrac{T}{2} - D -
\tfrac{D^2}{2}\bigr)/m + O(T^4/m^2)$ with $D = 2r - T$ (the
$R$--$U$ imbalance; the term has mean zero, as it must), whence
\[
c_1(\alpha) \;=\; \tfrac12 \lim_{T \to \infty}
E\Bigl[\mathbf{1}_{\{\text{surv} \leq T\}}
\Bigl(\tfrac{T}{2} - D_T - \tfrac{D_T^2}{2}\Bigr)\Bigr]
\;=\; \tfrac12\,E\Bigl[D_R + \tfrac{D_R^2 - R}{2};\ \text{ruin}\Bigr],
\]
the second form by optional stopping at the ruin time $R$ (both
$D_t$ and $D_t^2 - t$ are martingales vanishing at $0$; uniform
integrability holds by the exponential ruin-tail).
For $\alpha = 3/2$ an exact dynamic program gives
$c_1 = 0.622225\ldots$, matching the independently measured
$n\cdot(\text{normalized count} - \sigma) \to 0.622$.
A closed form would follow from the joint ruin-time/position
generating function of the periodic chain; we leave this open.
\end{remark}

\begin{proposition}[Nonsingularity of the boundary system]
\label{prop:nonsing}
For every rational $p/q$, the $q \times q$ boundary system
of~\cite{bridge} (and its sharpened variant of
Definition~\ref{def:sharp} below) is nonsingular, and its solution is
the probabilistic ruin function.
\end{proposition}

\begin{proof}
The space $V$ of solutions of the interior recursion that decay as
$s \to \infty$ has dimension $q$: it is the stable subspace of the
associated linear transfer cocycle, spanned by the modes
$A_j^{(i)} t_i^s$ of the $q$ sub-unit master-equation roots (with
polynomial corrections at multiple roots), whose count with
multiplicity is $q$ by~\cite[Prop.~5.3]{bridge}.
The true ruin function $\rho$ lies in $V$ (ruin probabilities tend to
$0$ as $s \to \infty$ by~\eqref{eq:linbound} and the strong law) and
attains the boundary values $1$.
Suppose $h \in V$ vanishes at all $q$ absorbing boundary states.
Extend $h$ by $0$ to the absorbing states; interior harmonicity makes
$h(s_{t \wedge \text{abs}}, j_{t \wedge \text{abs}})$ a bounded
martingale.
Almost surely the walk is either absorbed (where $h = 0$) or
$s_t \to \infty$ (where $h \to 0$ by decay), so dominated convergence
gives $h \equiv 0$.
Hence the evaluation map $V \to \mathbb{R}^q$ at the boundary is
injective, therefore bijective: the system is nonsingular and its
unique solution is $\rho$.
The same argument applies verbatim to the sharpened barrier.
\end{proof}

\section{Two coupling lemmas}

\begin{lemma}[Column comparison]\label{lem:columns}
Let $\alpha = p/q$ in lowest terms and $\varepsilon > 0$.
\begin{enumerate}
\item[(a)] For every $m \geq 1$:
  $\floor{(\alpha - \varepsilon)m} \leq \floor{\alpha m} - [\,q \mid m\,]$,
  with equality whenever $\varepsilon m < 1/q$.
\item[(b)] For every $m$ with $\varepsilon m < 1/q$:
  $\floor{(\alpha + \varepsilon)m} = \floor{\alpha m}$.
\item[(c)] For irrational $\alpha$ and any $x$ with $|x - \alpha| =
  \varepsilon$: $\floor{x m} = \floor{\alpha m}$ for all
  $m < m_*(\varepsilon) := \min\{m \geq 1 : \dnorm{\alpha m} \leq
  \varepsilon m\}$, where $\dnorm{\cdot}$ is the distance to the nearest
  integer; $m_*(\varepsilon) \to \infty$ as $\varepsilon \to 0$.
\end{enumerate}
\end{lemma}

\begin{proof}
(a) The fractional part of $\alpha m$ lies in
$\{0, 1/q, \ldots, (q-1)/q\}$ and vanishes exactly when $q \mid m$.
Subtracting $\varepsilon m > 0$ from an integer drops the floor by at
least one; subtracting less than $1/q$ from a number with fractional
part $\geq 1/q$ leaves the floor unchanged.
(b) Adding $\varepsilon m < 1/q$ cannot push the fractional part
($\leq (q-1)/q$) past $1$.
(c) Immediate from the definition; $m_* \to \infty$ because
$\dnorm{\alpha m} > 0$ for each fixed $m$.
\end{proof}

\begin{lemma}[Late-contact bound, free walk]\label{lem:chernoff}
Fix $\alpha > 1$ and $\theta \in (0, \theta_0(\alpha))$, where
$\theta_0$ is the positive root of
$\lambda_\theta := \tfrac12(e^{-\theta\alpha} + e^{\theta}) = 1$.
Then for every $c \geq 0$ and $T \geq 1$,
\[
P\bigl(\exists\, t \geq T:\ s_t \leq c\bigr)
\;\leq\; \frac{e^{\theta(c - \alpha + 1)}}{1 - \lambda_\theta}\;
\lambda_\theta^{\,T}.
\]
The same bound holds for any gap process whose allowance differs from
\eqref{eq:gap} by a bounded amount $b$, with $c$ replaced by $c + b$.
\end{lemma}

\begin{proof}
By~\eqref{eq:linbound}, $s_t \leq c$ implies
$Y_t := \alpha\tau_t - u_t \leq c - \alpha + 1$; $Y_t$ is an i.i.d.\
sum with increments $+\alpha$ or $-1$, each with probability
$\tfrac12$, and $E\,e^{-\theta\zeta} = \lambda_\theta < 1$ for small
$\theta > 0$ since
$\frac{d}{d\theta}\lambda_\theta|_{0} = \tfrac12(1 - \alpha) < 0$.
Markov's inequality and the geometric series over $t \geq T$ give the
claim.
\end{proof}

Since every column (right-step) consumes at least one coin flip,
``after column $M$'' implies $t \geq M$, and
Lemma~\ref{lem:chernoff} applies with $T = M$.

\section{Left limit: the sharpened-barrier system}

\begin{definition}\label{def:sharp}
For $\alpha = p/q$, the \emph{sharpened system} is the periodic walk
with gap process
$\tilde s_t = \floor{\alpha(\tau_t+1)} - [\,q \mid (\tau_t+1)\,] - u_t$,
i.e.\ the original walk with the absorbing barrier raised by one unit
in the columns $q \mid m$ (phase $0$).
Write $\tilde\sigma(p/q)$ for its survival probability and
$\tilde\rho(s, j)$ for its ruin probabilities.
\end{definition}

\begin{proposition}\label{prop:sharp-solve}
$\tilde\rho$ satisfies the same interior recursion as $\rho$, with the
same sub-unit master-equation roots $t_i$ and phase amplitudes
$A_j^{(i)}$ of~\cite{bridge}; the boundary system changes in exactly
one row: the phase-$0$ condition $\sum_i c_i A_0^{(i)}/t_i = 1$ is
replaced by $\sum_i c_i A_0^{(i)} = 1$ (absorption at level $0$
instead of $-1$).
Consequently $\tilde\sigma(p/q) = 1 - \tilde\rho(s_0, j_0)$ is
algebraic.
For $q = 1$ the system collapses to $\tilde\rho(s) = \rho^{\,s}$.
\end{proposition}

\begin{proof}
The interior recursion (conditioning on one coin flip) is untouched by
the barrier modification, so the exponential ansatz and the master
equation are unchanged.
The absorbing states are $s = -1$ at phases $j \neq 0$ and $s = 0$ at
phase $0$, giving the stated $q$ boundary rows; nonsingularity and the
probabilistic identification follow from
Proposition~\ref{prop:nonsing}.
For $q = 1$: the single condition $\tilde\rho(0) = 1$ with
$\tilde\rho(s) = c\,t^s$, $t = \rho$, forces $c = 1$.
\end{proof}

\begin{theorem}[Left limit]\label{thm:left}
For every rational $p/q$ with $p > q \geq 1$, $\gcd(p,q) = 1$:
\[
\lim_{\varepsilon \to 0^+} \sigma(p/q - \varepsilon)
= \tilde\sigma(p/q),
\qquad\text{with}\qquad
0 \leq \tilde\sigma - \sigma(p/q - \varepsilon)
\leq A\,\lambda^{\,1/(q\varepsilon)}
\]
for explicit $A, \lambda < 1$ depending only on $p/q$.
Hence $C^-(p/q) = \tfrac12\bigl(1 - \tilde\rho(s_0, j_0)\bigr)$ and
the jump
$J(p/q) = \tfrac12\bigl(\tilde\rho(s_0,j_0) - \rho(s_0,j_0)\bigr)$ is
algebraic.
\end{theorem}

\begin{proof}
Couple all walks on the same coin flips.
By Lemma~\ref{lem:columns}(a), $s^{(\varepsilon)}_t \leq \tilde s_t$
pointwise, so $\sigma(p/q - \varepsilon) \leq \tilde\sigma$; the left
limit (which exists by monotonicity of $\sigma$ and is independent of
the approach) is $\leq \tilde\sigma$.

For the reverse inequality, set $M = \floor{1/(q\varepsilon)}$.
By Lemma~\ref{lem:columns}(a) the two gap processes coincide while
$\tau_t < M$.
On the event $\{\tilde s \text{ survives}\} \setminus
\{s^{(\varepsilon)} \text{ survives}\}$, ruin of $s^{(\varepsilon)}$
occurs at some time $t$ with $\tau_t \geq M$, at which moment
$\tilde s_t \leq \varepsilon \tau_t + 1$
(the two allowances differ by at most $\varepsilon m + 1$ at column
$m$).
Partition the columns into dyadic blocks $\tau \in [2^i M, 2^{i+1} M)$:
on block $i$ the defect is at most $2^{i+1}\varepsilon M + 1 \leq
2^{i+1}/q + 1$, while $t \geq 2^i M$.
Lemma~\ref{lem:chernoff} bounds the probability of block-$i$ contact by
$B\,e^{\theta(2^{i+1}/q + 1)}\lambda_\theta^{\,2^i M}$.
Choosing $\theta$ with $e^{2\theta/q} \lambda_\theta^{M} < 1$ (possible
for $M \geq M_0(\theta)$), the sum over $i$ is dominated by a constant
multiple of its first term, giving the bound
$A \lambda_\theta^{\,M}$.
\end{proof}

\begin{corollary}[Integer slopes]\label{cor:integer}
$1 - 2\,C^-(k) = \rho_k^{\,k}$, i.e.\
$C^-(k) = (\tau_k - 1)/2$ and
$J(k) = (\tau_k - 1)/(2\tau_k^{\,k+1}) \sim 2^{-(k+2)}$,
where $\tau_k = 1/\rho_k$ is the multinacci constant.
\end{corollary}

\begin{proof}
Proposition~\ref{prop:sharp-solve} with $q = 1$:
$\tilde\rho(s_0) = \rho^{\,k}$ since $s_0 = k$; then
$2 - \tau_k = \tau_k^{-k}$ (master equation) gives the stated forms.
\end{proof}

\section{Right flatness}

\begin{theorem}[Right flatness]\label{thm:right}
For every rational $p/q$ and $0 < \varepsilon < 1$:
\[
0 \;\leq\; \sigma(p/q + \varepsilon) - \sigma(p/q)
\;\leq\; A\,\lambda^{\,1/(q\varepsilon)} .
\]
Consequently all one-sided right derivatives of $C$ at $p/q$ vanish:
$C$ is flat to infinite order from the right.
\end{theorem}

\begin{proof}
Monotonicity gives the left inequality.
By Lemma~\ref{lem:columns}(b) the gap processes coincide while
$\tau_t < M = \floor{1/(q\varepsilon)}$, so on
$\{s^{(+\varepsilon)} \text{ survives}\} \setminus
\{s \text{ survives}\}$ the unperturbed walk is ruined at a column
$\geq M$ while carrying a bonus of at most $\varepsilon\tau + 1$;
the dyadic-block argument of Theorem~\ref{thm:left} applies verbatim
to the unperturbed walk's late contact, giving the bound.
Flatness: $A e^{-\kappa/\varepsilon}/\varepsilon^n \to 0$ for every $n$,
with $\kappa = |{\ln \lambda}|/q$.
\end{proof}

\begin{remark}
The same argument applied below the jump gives
$\tilde\sigma - \sigma(p/q - \varepsilon) \leq A\lambda^{1/(q\varepsilon)}$
(Theorem~\ref{thm:left}): the approach to the \emph{left limit} is also
exponentially fast.
Near each rational, $C$ is therefore a two-level step function up to
exponentially small corrections --- consistent with the geometric
convergence (in $K$) of the continued-fraction ladders observed
numerically.
\end{remark}

\section{Continuity at irrationals}

\begin{theorem}\label{thm:irrational}
$\sigma$ (hence $C$) is continuous at every irrational $\alpha$, with
\[
|\sigma(x) - \sigma(\alpha)| \;\leq\; A\,\lambda^{\,m_*(|x - \alpha|)},
\]
$m_*$ as in Lemma~\ref{lem:columns}(c).
If $\alpha$ has Diophantine type $\kappa$ (i.e.\
$\dnorm{\alpha m} \geq c\,m^{1-\kappa}$), then
$m_*(\varepsilon) \geq (c/\varepsilon)^{1/\kappa}$ and the modulus is
stretched-exponential; for bounded type (e.g.\ quadratic irrationals),
$|\sigma(x) - \sigma(\alpha)| \leq A\,\lambda^{\,c/\sqrt{\varepsilon}}$.
\end{theorem}

\begin{proof}
Identical coupling: the words agree up to column $m_*$, the defect/bonus
beyond column $m$ is at most $\varepsilon m + 1$, and the dyadic
late-contact argument gives the bound.
\end{proof}

\section{The pure-jump structure}

We now show that the jumps are \emph{all} there is: the
Lebesgue--Stieltjes measure of $C$ has no continuous component.
The proof combines the two-sided exponential flatness with Dirichlet's
approximation theorem; the decisive cancellation is that at scale
$\delta = 1/(qQ)$ the flatness exponent $1/(q\delta)$ equals $Q$,
\emph{uniformly in the denominator} $q$.

\begin{lemma}[Uniform flatness constants]\label{lem:uniform}
Let $[\alpha_0, \beta_0] \subset (1, \infty)$ be compact.
There exist $A < \infty$ and $\lambda < 1$ such that for every rational
$p/q \in [\alpha_0, \beta_0]$ and every $\varepsilon > 0$,
\[
C(p/q + \varepsilon) - C(p/q) \;\leq\; A\,\lambda^{1/(q\varepsilon)},
\qquad
C^-(p/q) - C(p/q - \varepsilon) \;\leq\; A\,\lambda^{1/(q\varepsilon)} .
\]
\end{lemma}

\begin{proof}
Inspect the proofs of Theorems~\ref{thm:left} and~\ref{thm:right} with
$M = \floor{1/(q\varepsilon)}$.
Fix $\theta \in (0, \theta_0(\alpha_0))$ once for the whole interval;
since $\lambda_\theta(\alpha) = \tfrac12(e^{-\theta\alpha} + e^\theta)$
is decreasing in $\alpha$, the value
$\lambda := \lambda_\theta(\alpha_0) < 1$ bounds it for all slopes in
the interval, and the prefactor of Lemma~\ref{lem:chernoff} is bounded
by $B = e^{\theta}/(1-\lambda)$ uniformly (slopes are bounded).
In the dyadic-block sum the block-$i$ defect is at most
$2^{i+1}\varepsilon M + 1 \leq 2^{i+1}/q + 1 \leq 2^{i+1} + 1$, so the
sum is dominated by
$B e^{\theta} \sum_i (e^{2\theta} \lambda^{M})^{2^i}
\leq 2 B e^{\theta} e^{2\theta} \lambda^{M}$
whenever $e^{2\theta}\lambda^{M} \leq \tfrac12$, i.e.\ for
$M \geq M_0(\theta, \lambda)$ --- a uniform threshold.
For $M < M_0$ the left-hand sides are at most the total variation
$\tfrac12$, absorbed by enlarging $A$ to
$\max(2Be^{3\theta}, \tfrac12 \lambda^{-M_0})$.
\end{proof}

\begin{theorem}[Pure jump structure]\label{thm:purejump}
The continuous part of the Lebesgue--Stieltjes measure of $C$ vanishes
on $(1, \infty)$: for all $1 < \alpha < \beta$,
\[
C(\beta) - C(\alpha) \;=\; \sum_{r \in (\alpha, \beta] \cap \mathbb{Q}}
J(r) .
\]
In particular $C' = 0$ Lebesgue-almost everywhere, and $C$ is the
distribution function of the purely atomic measure
$\sum_{p/q} J(p/q)\,\delta_{p/q}$ with the algebraic weights of
Theorem~\ref{thm:left}.
\end{theorem}

\begin{proof}
Write $\mu_C = \mu_{\mathrm{atom}} + \mu_{\mathrm{cont}}$ with
$\mu_{\mathrm{atom}} = \sum_r J(r)\delta_r$ ($C$ is monotone and
right-continuous, so this is the usual decomposition; the atoms sit at
the rationals by Theorems~\ref{thm:left}--\ref{thm:irrational}).
Fix a compact $[\alpha_0, \beta_0] \subset (1,\infty)$ and let
$A, \lambda$ be as in Lemma~\ref{lem:uniform}.

Let $Q \geq 1$.
By Dirichlet's approximation theorem, every $x \in [\alpha_0, \beta_0]$
satisfies $|x - p/q| < 1/(qQ)$ for some rational $p/q$ with
$q \leq Q$; hence the intervals
$I_{p/q} = \bigl(p/q - \tfrac{1}{qQ},\; p/q + \tfrac{1}{qQ}\bigr)$,
taken over rationals in $[\alpha_0 - 1, \beta_0 + 1]$ with $q \leq Q$,
cover $[\alpha_0, \beta_0]$.
There are at most $(\beta_0 - \alpha_0 + 2)\sum_{q \leq Q} q
\leq c\,Q^2$ of them.

Since $\mu_{\mathrm{cont}}$ has no atoms,
\[
\mu_{\mathrm{cont}}(I_{p/q})
= \mu_{\mathrm{cont}}\bigl(I_{p/q} \setminus \{p/q\}\bigr)
\leq \Bigl[C\bigl(\tfrac{p}{q} + \tfrac{1}{qQ}\bigr) - C\bigl(\tfrac{p}{q}\bigr)\Bigr]
+ \Bigl[C^-\bigl(\tfrac{p}{q}\bigr) - C\bigl(\tfrac{p}{q} - \tfrac{1}{qQ}\bigr)\Bigr]
\;\leq\; 2A\,\lambda^{Q},
\]
where the final step uses Lemma~\ref{lem:uniform} with
$\varepsilon = 1/(qQ)$, so that $1/(q\varepsilon) = Q$ ---
\emph{independently of} $q$.
Therefore
\[
\mu_{\mathrm{cont}}\bigl([\alpha_0, \beta_0]\bigr)
\;\leq\; c\,Q^2 \cdot 2A\,\lambda^{Q}
\;\xrightarrow[Q \to \infty]{}\; 0 .
\]
Exhausting $(1,\infty)$ by compacts gives $\mu_{\mathrm{cont}} = 0$.
The a.e.\ vanishing derivative follows since a monotone pure-jump
function has $C' = 0$ off the (countable) atom set union a Lebesgue-null
set.
\end{proof}

\begin{remark}
The numerical sum rule of the companion session is the quantitative
shadow of Theorem~\ref{thm:purejump}: on $(27/20, 29/20]$ the exact
jumps with $q \leq 60$ account for $95.10\%$ of $\Delta C$, and the
extrapolated tail matches the remainder to $6.6 \times 10^{-6}$.
Note that flatness alone does not imply the theorem --- Minkowski's
question-mark function is flat at every rational yet continuous; the
covering argument needs the \emph{quantitative} uniform rate of
Lemma~\ref{lem:uniform}.
\end{remark}

\section{The exact decay rate of the jumps}

\begin{theorem}[Contact rate]\label{thm:rate}
For $\alpha > 1$ let
\[
\mu(\alpha) \;=\; \inf_{0 < e^\theta < 2}
\frac{e^{-\theta\alpha}}{2 - e^{\theta}}
\;=\; \frac{(\alpha+1)^{\alpha+1}}{2^{\alpha+1}\,\alpha^{\alpha}},
\qquad e^{\theta^*} = \frac{2\alpha}{\alpha + 1},
\]
and for $c_0 \geq 0$ let
$K_m = \{\exists\, t:\ \tau_t = m,\ s_t \leq c_0\}$ be the event that
the gap drops to $c_0$ or below while in column $m$.
Then
\[
P(K_m) \;\leq\; e^{\theta^*(c_0 + 1)}\,\mu(\alpha)^{\,m+1},
\qquad
P\bigl(K_m \cap \{\text{survival}\}\bigr) \;\geq\;
\mu(\alpha)^{\,m}\, e^{-c\sqrt{m}} \quad (m \geq m_0),
\]
so $\lim_m \frac{1}{m}\log P(K_m \cap \{\text{survival}\}) = \log
\mu(\alpha)$.
\end{theorem}

\begin{proof}
\emph{Upper bound.}
The up-steps between consecutive right-steps are i.i.d.\
$\mathrm{Geom}(\tfrac12)$ variables $G_j$ ($P(G = g) = 2^{-g-1}$,
$E e^{\theta G} = 1/(2 - e^\theta)$ for $e^\theta < 2$).
The gap while in column $m$ is $\floor{\alpha(m+1)} - u$, so $K_m$
forces $U_{m+1} := G_1 + \cdots + G_{m+1} \geq \alpha(m+1) - 1 - c_0$.
Markov's inequality with $e^{\theta U}$ gives
$P(K_m) \leq e^{\theta(c_0 + 1)}\bigl[e^{-\theta\alpha}/(2 -
e^\theta)\bigr]^{m+1}$; optimize $\theta = \theta^*$.

\emph{Lower bound.}
Let $Q$ retile the first $m+1$ inter-arrival variables as
$\mathrm{Geom}$ with mean $\alpha$ (success probability
$1/(1+\alpha)$ --- precisely the $\theta^*$-tilt of
$\mathrm{Geom}(\tfrac12)$), leaving the rest of the randomness
untouched.
On any event where $U_{m+1} \leq \alpha(m+1) + K\sqrt{m}$ the
Radon--Nikodym derivative satisfies
$dP/dQ \geq \mu(\alpha)^{m+1} e^{-\theta^* K\sqrt{m}}$.
Under $Q$ the column increments $r_j - G_j$ have mean zero and finite
variance, and we exhibit a $Q$-scenario of probability
$\geq e^{-c'\sqrt{m}}$ inside $K_m \cap \{U \leq \alpha(m+1) +
K\sqrt{m}\}$ with survival up to column $m$:
force $G_j = 0$ for the first $k_0 = \lceil c_1\sqrt{m}\rceil$ columns
($Q$-cost $(1+\alpha)^{-k_0}$), lifting the gap to height
$\asymp \sqrt{m}$; by Kolmogorov's inequality the zero-drift middle
section stays positive with probability $\geq \tfrac12$ for a suitable
$c_1$; finally force $\asymp \sqrt{m}$ heavy columns at the end to
bring the gap back below $c_0$ at column $m$ ($Q$-cost again
$e^{-c''\sqrt{m}}$).
After column $m$ the original measure resumes; survival from a
bounded gap costs a further constant factor
$\sigma_{\min}(\alpha) > 0$ (force a bounded number of right-steps,
then Lemma~\ref{lem:chernoff}).
Combining, $P(K_m \cap \text{survival}) \geq
E_Q[\mathbf{1}\,dP/dQ]\,\sigma_{\min} \geq \mu^{m} e^{-c\sqrt{m}}$.
\end{proof}

\begin{corollary}[Jump decay]\label{cor:jumpdecay}
For every rational $p/q$ in a compact slope interval,
\[
c_1\, e^{-c\sqrt{q}}\;\mu(p/q)^{\,q}
\;\leq\; J(p/q) \;\leq\;
c_2\, \mu(p/q)^{\,q},
\]
i.e.\ $J(p/q) = \mu(p/q)^{\,q\,(1 + o(1))}$.
In particular the jump series $\sum J$ converges geometrically in $q$
and the pre-asymptotic plateau $J \approx c/q^2$ observed for small $q$
must cross over to exponential decay (numerically at
$q^* \sim (\alpha - 1)^{-2}$).
\end{corollary}

\begin{proof}
By the renewal formula (a one-row Sherman--Morrison perturbation of
the boundary system of~\cite{bridge}, cf.\
Proposition~\ref{prop:sharp-solve}),
$J = H_0 (1 - \rho(0,0))/(2 h(0,0))$, where $H_0$ is the probability
of absorption at the phase-$0$ boundary state, $h(0,0) \in
[\tfrac12, 1]$, and $1 - \rho(0,0) \in (\sigma_{\min}, 1)$.
Absorption at phase $0$ requires the gap to reach $-1$ in a column
divisible by $q$, the first opportunity being column $q$; so
$H_0 \leq \sum_{j \geq 1} P(K_{jq}) \leq B \mu^{q}$ by
Theorem~\ref{thm:rate} (upper), and
$H_0 \geq P(K_q \cap \text{ruin in column } q)
\geq \mu^{q} e^{-c\sqrt{q}}$ by the lower construction with a final
forced dip below $-1$ within column $q$.
\end{proof}

The closed form of Theorem~\ref{thm:left} was tested against one-sided
ladder extrapolations at 25 rationals (all slopes measured in the
companion session); it agrees within each measurement's accuracy,
down to $6 \times 10^{-11}$ at $8/5$ and $4 \times 10^{-9}$ at $5/3$.
Exact example: for $3/2$ (sub-unit roots $0.43691$, $0.71974$, both
real),
\[
J(3/2) = 0.0272922840\ldots,
\qquad
64x^6 - 896x^5 + 4336x^4 - 7688x^3 + 2100x^2 - 528x + 13 = 0 .
\]

The exponential rates of Theorems~\ref{thm:left}--\ref{thm:right} are
visible as geometric convergence of the CF-extension ladders
$x_K \to p/q$ (where $\varepsilon_K = 1/(q\,q_K)$, so
$\lambda^{1/(q\varepsilon)} = \lambda^{q_K}$): measured per-column
contraction $\mu$ versus the per-flip Chernoff bound
$\lambda_{\theta^*}$:

\midrule
$3/2$ & $0.929$--$0.934$ & $0.9801$ \\

\smallskip\noindent
(The measured rates drift slowly upward with $K$, suggesting a nearby

The asymptotic contraction rates match the Cram\'er rate of
Theorem~\ref{thm:rate}: after the empirical $q^{-3/2}$ prefactor
correction, the ladders give $0.9532$ vs $\mu(3/2) = 0.95090$,

(all within $0.3\%$), and the exact jumps along the $\sqrt2$-convergent
slopes $17/12 \to 239/169$ give
$\mu_{\mathrm{emp}} = 0.96466$ vs $\mu(\sqrt2) = 0.964920$
(within $2.7 \times 10^{-4}$).
The sharp polynomial prefactor ($q^{-3/2}$ empirically) is not proven
here.

\begin{remark}[Status]
With Theorem~\ref{thm:zero} (path-count identification) and
Proposition~\ref{prop:nonsing} (nonsingularity), the chain
\emph{path counts $\to$ survival probabilities $\to$ boundary systems
$\to$ one-sided regularity, closed-form jumps, pure-jump structure and
exact decay rate} is self-contained.
The numerical agreement tables (here and in~\cite{bridge}) serve as
implementation checks, no longer as evidence for the identification.
Open: the sharp polynomial prefactor in Theorem~\ref{thm:rate}
(empirically $q^{-3/2}$), and a closed form for the finite-size
constant $c_1(\alpha)$ of Remark~\ref{rem:finitesize}.
\end{remark}
